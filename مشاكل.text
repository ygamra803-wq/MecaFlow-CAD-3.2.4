قمت بمراجعة كاملة لنظام **MecaFlow-CAD V2.3** (Zero-Gap Laser CAD Pro) والذي هو نظام متخصص في التصميم الهندسي لأنابيب المطبخ ومقاطع الليزر. فيما يلي **قائمة مفصلة بالمشاكل التقنية، مشاكل الكود، ومشاكل المنطق** التي اكتشفتها خلال المراجعة، مع الحلول المقترحة لكل منها.

export class CSGProcessor {
  private trackedBrushes: WeakSet<Brush> = new WeakSet();
  private trackedGeometries: WeakSet<THREE.BufferGeometry> = new WeakSet();
  private evaluator: Evaluator;

  // إضافة mechanism للتخلص التلقائي
  cleanupUnused() {
    // استخدام WeakSet لتجنب memory leaks
    const brushesToDispose = Array.from(this.trackedBrushes);
    brushesToDispose.forEach(brush => {
      if (brush.geometry) brush.geometry.dispose();
      if (brush.material) {
        if (Array.isArray(brush.material)) {
          brush.material.forEach(m => m.dispose());
        } else {
          brush.material.dispose();
        }
      }
    });
    this.trackedBrushes = new WeakSet();
  }
}

// تحسين ErrorBoundary
class EnhancedErrorBoundary extends React.Component<{
  children: React.ReactNode;
  onError?: (error: Error, componentStack: string) => void;
  fallback?: React.ReactNode;
}, { hasError: boolean; error: Error | null }> {

  static getDerivedStateFromError(error: Error) {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
    // تسجيل الخطأ في النظام المركزي
    ErrorHandler.handle(error, {
      component: errorInfo.componentStack,
      location: window.location.href
    });

    // إشعار النظام المركزي
    if (this.props.onError) {
      this.props.onError(error, errorInfo.componentStack);
    }

    // إعادة تحميل التطبيق في حالة أخطاء حرجة
    if (error.message.includes('WebGL') ||
        error.message.includes('out of memory')) {
      window.location.reload();
    }
  }

  render() {
    if (this.state.hasError) {
      return this.props.fallback || (
        <ErrorFallback
          error={this.state.error}
          onRetry={() => this.setState({ hasError: false, error: null })}
        />
      );
    }
    return this.props.children;
  }
}
```

// تحسين useProjectHistory
export function useProjectHistory(projectId?: string) {
  const [history, setHistory] = useState<GeometryState[]>([]);
  const [currentIndex, setCurrentIndex] = useState<number>(-1);
  const MAX_HISTORY_SIZE = 50; // حد أقصى للـ history

  const updateGeometry = useCallback((newGeometry: GeometryState) => {
    setHistory(prev => {
      const newHistory = prev.slice(0, currentIndex + 1);

      // إذا كان التغير صغيرا، دمج مع آخر operation
      if (newHistory.length > 0) {
        const last = newHistory[newHistory.length - 1];
        if (isSimilarGeometry(last, newGeometry)) {
          return newHistory; // تجاهل التغير الصغير
        }
      }

      // إضافة التغير الجديد
      const updated = [...newHistory, newGeometry];

      // تقليل حجم history إذا تجاوز الحد
      if (updated.length > MAX_HISTORY_SIZE) {
        return updated.slice(-MAX_HISTORY_SIZE);
      }

      return updated;
    });
    setCurrentIndex(prev => prev + 1);
  }, [currentIndex]);

  // دالة لمقارنة geometries
  const isSimilarGeometry = (a: GeometryState, b: GeometryState): boolean => {
    // مقارنة بسيطة للـ config
    return JSON.stringify(a) === JSON.stringify(b);
  };
}
```

  if (!mesh) {

      'Cannot export null/undefined mesh to STEP',
      'error'

export class ExportService {

class PatchManager:

            return True

  static registerHotkey(key: string, callback: () => void) {
    this.handlers.set(key, callback);
  }

  static unregisterHotkey(key: string) {
    this.handlers.delete(key);
  }

  static setupGlobalListener() {
    document.addEventListener('keydown', (e) => {
      // تجاهل إذا كان هناك input نشط
      if (e.target instanceof HTMLInputElement ||
          e.target instanceof HTMLTextAreaElement) {
        return;
      }

      const key = this.getKeyCombination(e);
      const handler = this.handlers.get(key);

      if (handler) {
        e.preventDefault();
        handler();
      }
    });
  }

  private static getKeyCombination(e: KeyboardEvent): string {
    const parts: string[] = [];

    if (e.ctrlKey) parts.push('Ctrl');
    if (e.shiftKey) parts.push('Shift');
    if (e.altKey) parts.push('Alt');
    if (e.metaKey) parts.push('Meta');

    parts.push(e.key);

    return parts.join('+');
  }
}

// 1. إضافة sanitization للـ inputs
export class InputSanitizer {
  static sanitizeString(input: string): string {
    // إزالة HTML tags
    return input.replace(/<[^>]*>/g, '');
  }

  static sanitizeNumber(input: string): number | null {
    const num = parseFloat(input);
    if (isNaN(num)) return null;

    // التحقق من أن الرقم في نطاق معقول
    if (Math.abs(num) > 1e10) return null;

    return num;
  }

  static sanitizeObject(obj: unknown): Record<string, unknown> {
    if (typeof obj !== 'object' || obj === null) {
      return {};
    }

    const result: Record<string, unknown> = {};

    for (const [key, value] of Object.entries(obj)) {
      if (typeof value === 'string') {
        result[key] = this.sanitizeString(value);
      } else if (typeof value === 'number') {
        result[key] = this.sanitizeNumber(value.toString());
      } else if (typeof value === 'object' && value !== null) {
        result[key] = this.sanitizeObject(value);
      } else {
        result[key] = value;
      }
    }

    return result;
  }
}

// إضافة Rate Limiter
class RateLimiter {
  private requests: Map<string, { count: number; lastRequest: number }> = new Map();
  private maxRequests: number;
  private windowMs: number;

  constructor(maxRequests: number = 10, windowMs: number = 1000) {
    this.maxRequests = maxRequests;
    this.windowMs = windowMs;
  }

  check(ip: string): boolean {
    const now = Date.now();
    const record = this.requests.get(ip);

    if (!record) {
      this.requests.set(ip, { count: 1, lastRequest: now });
      return true;
    }

    // إذا انقضت الفترة، إعادة التعديل
    if (now - record.lastRequest > this.windowMs) {
      this.requests.set(ip, { count: 1, lastRequest: now });
      return true;
    }

    // إذا تجاوز الحد
    if (record.count >= this.maxRequests) {
      return false;
    }

    // زيادة العداد
    this.requests.set(ip, { count: record.count + 1, lastRequest: now });
    return true;
  }

  reset(ip: string) {
    this.requests.delete(ip);
  }
}

      'RATE_LIMITED',
      'Too many requests. Please try again later.',

    );

enum LogLevel {
  DEBUG = 'debug',
  INFO = 'info',
  WARN = 'warn',
  ERROR = 'error',
  CRITICAL = 'critical'
}

interface LogEntry {
  level: LogLevel;
  message: string;
  context?: Record<string, unknown>;
  timestamp: string;
  stack?: string;
}

class Logger {
  private static logs: LogEntry[] = [];
  private static maxLogs = 1000;
  private static listeners: Array<(entry: LogEntry) => void> = [];

  static log(
    level: LogLevel,
    message: string,
    context?: Record<string, unknown>,
    stack?: string
  ) {
    const entry: LogEntry = {
      level,
      message,
      context,
      timestamp: new Date().toISOString(),
      stack
    };

    this.logs.push(entry);

    // الحفاظ على حجم logs
    if (this.logs.length > this.maxLogs) {
      this.logs.shift();
    }

    // طباعة إلى console
    const method = level === LogLevel.ERROR || level === LogLevel.CRITICAL
      ? console.error
      : level === LogLevel.WARN
        ? console.warn
        : console.log;

    method(`[${level.toUpperCase()}]`, message, context);

    // إشعار listeners
    this.listeners.forEach(listener => listener(entry));

    // إرسال إلى server في حالة errors
    if (level === LogLevel.ERROR || level === LogLevel.CRITICAL) {
      this.sendToServer(entry);
    }
  }

  static debug(message: string, context?: Record<string, unknown>) {
    this.log(LogLevel.DEBUG, message, context);
  }

  static info(message: string, context?: Record<string, unknown>) {
    this.log(LogLevel.INFO, message, context);
  }

  static warn(message: string, context?: Record<string, unknown>) {
    this.log(LogLevel.WARN, message, context);
  }

  static error(message: string, context?: Record<string, unknown>, stack?: string) {
    this.log(LogLevel.ERROR, message, context, stack);
  }

  static critical(message: string, context?: Record<string, unknown>, stack?: string) {
    this.log(LogLevel.CRITICAL, message, context, stack);
  }

  static subscribe(listener: (entry: LogEntry) => void) {
    this.listeners.push(listener);
  }

  static getLogs(): LogEntry[] {
    return [...this.logs];
  }

  static clearLogs() {
    this.logs = [];
  }

  private static async sendToServer(entry: LogEntry) {
    try {
      await fetch('/api/logs', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(entry)
      });
    } catch (e) {
      console.error('Failed to send logs to server:', e);
    }
  }
}

      return false;

      root.style.setProperty(`--color-${key}`, value);
    });

    // تطبيق الخطوط
    Object.entries(this.currentTheme.fonts).forEach(([key, value]) => {
      root.style.setProperty(`--font-${key}`, value);
    });

    // تطبيق الاتجاه
    root.dir = this.currentTheme.direction;
    root.lang = this.currentTheme.direction === 'rtl' ? 'ar' : 'en';
  }
}

// الاستخدام:
const themeManager = new ThemeManager();

// تغيير theme
themeManager.setTheme('arabic');

// الاستماع للتغييرات
themeManager.subscribe((theme) => {
  console.log(`Theme changed to ${theme.name}`);
});
